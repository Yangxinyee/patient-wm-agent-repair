    mu = x.mean(-1, keepdims=True); var = x.var(-1, keepdims=True)

def actf(x, kind="relu"):
    if kind == "relu": return np.maximum(x, 0)
    if kind == "gelu":
        from scipy.special import erf
        return 0.5 * x * (1 + erf(x / np.sqrt(2)))
